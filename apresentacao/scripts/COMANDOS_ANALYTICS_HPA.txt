# BLOCO 6 — TERMINAL 1: EVENTOS DO LOTE NO DYNAMODB      (FALA 6)
# ============================================================
# Rodar depois que a fila zerar (terminal 2 com fila=0).
if (-not $envio.RunId) { throw 'Execute o BLOCO 3 neste terminal primeiro.' }
$dadosJson = aws dynamodb scan --table-name ToggleMasterAnalytics --region us-east-1 --output json
if ($LASTEXITCODE -ne 0) { throw 'Falha ao consultar DynamoDB.' }
$dados = $dadosJson | ConvertFrom-Json
$eventos = @($dados.Items | Where-Object { $_.flag_name.S -eq $envio.RunId })
$eventos | Select-Object -First 10 | ForEach-Object {
    [pscustomobject]@{
        Evento = $_.event_id.S
        Lote = $_.flag_name.S
        Usuario = $_.user_id.S
        Resultado = $_.result.BOOL
        Data = $_.timestamp.S
    }
} | Format-Table -AutoSize
'Eventos encontrados neste lote: ' + $eventos.Count
